    const avatarHtml = (fullNm, size = 24, pid = null) => {
      const u = getPhoto(fullNm, pid);
      const last = String(fullNm || '').replace(/,\s*(jr|sr|ii|iii|iv|v)\.?$/i, '').split(' ').pop();
      const imgStr = u
        ? '<img src="' + esc(u) + '" referrerpolicy="no-referrer" onerror="this.style.display=\'none\'" style="width:' + size + 'px;height:' + size + 'px;border-radius:50%;object-fit:cover;object-position:top;border:1.5px solid var(--border);display:block;margin:0 auto;">'
        : '<div style="width:' + size + 'px;height:' + size + 'px;border-radius:50%;background:#ddd;display:flex;align-items:center;justify-content:center;font-size:8px;font-weight:700;margin:0 auto;">' + esc(last.slice(0,2)) + '</div>';
      return '<div style="text-align:center;">' + imgStr + '<div style="font-size:8px;color:var(--muted);line-height:1.1;margin-top:1px;">' + esc(last) + '</div></div>';
    };

    const tms = cachedTeamStats?.slices?.all || [];
    const lgMean = (key, fallback) => {
      const arr = tms.map(t => t[key]).filter(v => v != null);
      return arr.length ? (arr.reduce((a, b) => a + b, 0) / arr.length) : fallback;
    };
    const seasonPctRank = (val, key, lb) => {
      const arr = tms.map(t => t[key]).filter(v => v != null).sort((a, b) => a - b);
      if (!arr.length) return 50;
      const below = arr.filter(v => v <= val).length;
      const raw = Math.round((below / arr.length) * 100);
      return lb ? (100 - raw) : raw;
    };
    const normCdfPct = (z) => {
      const t = 1 / (1 + 0.2316419 * Math.abs(z));
      const d = 0.3989423 * Math.exp(-z * z / 2);
      const p = d * t * (0.3193815 + t * (-0.3565638 + t * (1.781478 + t * (-1.821256 + t * 1.330274))));
      const cdf = z > 0 ? (1 - p) : p;
      return Math.max(1, Math.min(99, Math.round(cdf * 100)));
    };
    const singleGamePct = (val, mean, sd, lb) => {
      const z = (val - mean) / sd;
      return normCdfPct(lb ? -z : z);
    };
    const cellBg = (pct) => {
      if (pct >= 75) return 'background:#b7ebbf;';
      if (pct >= 55) return 'background:#dcfce7;';
      if (pct >= 45) return 'background:#ffffff;';
      if (pct >= 25) return 'background:#fee2e2;';
      return 'background:#f5a3a3;';
    };
    const snapCell = (val, pct) => '<td style="padding:11px 3px 4px;border:1px solid #d4d4d4;position:relative;text-align:center;' + cellBg(pct) + '">' +
      '<div style="position:absolute;top:1px;right:2px;font-size:7.5px;line-height:1;color:#444;font-weight:600;">' + pct + '%</div>' +
      '<div style="font-size:11px;font-weight:700;color:#000;line-height:1.1;">' + (val * 100).toFixed(1) + '%</div></td>';

    const dreLinkHtml = '<a href="https://fansided.com" target="_blank" rel="noopener" title="Daily RAPM Estimate (Kevin Ferrigan, Nylon Calculus) — empirical box-score regression estimating per-100-possession RAPM impact" style="color:var(--blue);text-decoration:underline;">DRE</a>';

    if (_recapOffset > 0) {
      const fg = futureGames.at(_recapOffset - 1);
      const oppAb = String(fg.opp || 'OPP').toUpperCase();
      if (titleEl) titleEl.textContent = _recapOffset === 1 ? 'Next Game Preview' : 'Upcoming Game';

      const oppLogo = glLogoUrl(oppAb);
      const oppImgHtml = oppLogo ? ('<img src="' + oppLogo + '" title="' + esc(oppAb) + '" style="width:38px;height:38px;object-fit:contain;display:block;">') : ('<div style="font-family:Oswald,sans-serif;font-weight:700;font-size:13px;color:var(--muted);">' + esc(oppAb) + '</div>');

      const mxcRow = tms.find(t => String(t.TEAM_ABBREVIATION || '').toUpperCase() === 'MXC') || {};
      const oppRow = tms.find(t => String(t.TEAM_ABBREVIATION || '').toUpperCase() === oppAb) || {};

      const mEfg = Number(mxcRow.EFG_PCT || 0.53), mTov = Number(mxcRow.TM_TOV_PCT || 0.155), mOrb = Number(mxcRow.OREB_PCT || 0.285), mFtr = Number(mxcRow.FTA_RATE || 0.24);
      const oEfg = Number(oppRow.EFG_PCT || 0.53), oTov = Number(oppRow.TM_TOV_PCT || 0.155), oOrb = Number(oppRow.OREB_PCT || 0.285), oFtr = Number(oppRow.FTA_RATE || 0.24);

      const mxcCells = snapCell(mEfg, seasonPctRank(mEfg, 'EFG_PCT', false))
                     + snapCell(mTov, seasonPctRank(mTov, 'TM_TOV_PCT', true))
                     + snapCell(mOrb, seasonPctRank(mOrb, 'OREB_PCT', false))
                     + snapCell(mFtr, seasonPctRank(mFtr, 'FTA_RATE', false));

      const oppCells = snapCell(oEfg, seasonPctRank(oEfg, 'EFG_PCT', false))
                     + snapCell(oTov, seasonPctRank(oTov, 'TM_TOV_PCT', true))
                     + snapCell(oOrb, seasonPctRank(oOrb, 'OREB_PCT', false))
                     + snapCell(oFtr, seasonPctRank(oFtr, 'FTA_RATE', false));

      const allLeagueDuos = Array.isArray(llData?.duos) ? llData.duos : [];
      const oppDuosRaw = allLeagueDuos.filter(d => String(d.TEAM_ABBREVIATION || '').toUpperCase() === oppAb);
      const minOppPoss = [50, 35, 20, 10].find(t => oppDuosRaw.filter(d => Number(d.POSS || 0) >= t).length >= 2) || 5;
      const oppDuos = oppDuosRaw
        .filter(d => Number(d.POSS || 0) >= minOppPoss)
        .sort((a, b) => Number(b.NET_RATING || 0) - Number(a.NET_RATING || 0));

      const renderOppDuo = (title, dObj, isGood) => {
        if (!dObj) return '<div></div>';
        const col = isGood ? 'var(--green)' : 'var(--red)';
        const net = Number(dObj.NET_RATING || 0);
        const ids = String(dObj.GROUP_ID || '').split('-').filter(Boolean);
        const nms = String(dObj.GROUP_NAME || '').split(' - ');
        return '<div style="background:#f8fafc;border:1px solid var(--border);padding:3px 5px;min-width:0;">' +
          '<div style="display:flex;justify-content:space-between;align-items:baseline;font-size:8.5px;margin-bottom:2px;gap:2px;">' +
            '<span style="font-weight:700;color:#000;white-space:nowrap;">' + title + ' <span style="color:var(--muted);font-weight:400;">(' + Number(dObj.POSS || 0) + 'p)</span></span>' +
            '<span style="font-weight:700;color:' + col + ';white-space:nowrap;">' + (net >= 0 ? '+' : '') + net.toFixed(1) + '</span>' +
          '</div>' +
          '<div style="display:grid;grid-template-columns:repeat(2,1fr);gap:2px;">' +
            nms.map((nm, idx2) => avatarHtml(nm, 24, ids.at(idx2))).join('') +
          '</div></div>';
      };

      const oppBestD = oppDuos.at(0);
      const oppWorstD = oppDuos.length > 1 ? oppDuos.at(-1) : null;
      const oppDres = Array.isArray(lgDre?.[oppAb]) ? lgDre[oppAb] : [];
      const oHiDre = oppDres.at(0);
      const oLoDre = oppDres.length > 1 ? oppDres.at(-1) : null;

      const oHiHtml = oHiDre ? (
        '<div style="display:flex;align-items:center;gap:5px;background:#f8fafc;padding:3px 4px;border:1px solid var(--border);">' +
        avatarHtml(oHiDre.name, 26, oHiDre.id) +
        '<div style="font-size:9px;line-height:1.2;"><div style="font-weight:700;">High ' + dreLinkHtml + '</div>' +
        '<div style="color:var(--green);font-weight:700;font-size:10.5px;">' + (oHiDre.dre >= 0 ? '+' : '') + oHiDre.dre.toFixed(1) + '</div>' +
        '<div style="color:var(--muted);">' + oHiDre.poss + ' poss</div></div></div>'
      ) : '';

      const oLoHtml = oLoDre ? (
        '<div style="display:flex;align-items:center;gap:5px;background:#f8fafc;padding:3px 4px;border:1px solid var(--border);">' +
        avatarHtml(oLoDre.name, 26, oLoDre.id) +
        '<div style="font-size:9px;line-height:1.2;"><div style="font-weight:700;">Low ' + dreLinkHtml + '</div>' +
        '<div style="color:var(--red);font-weight:700;font-size:10.5px;">' + (oLoDre.dre >= 0 ? '+' : '') + oLoDre.dre.toFixed(1) + '</div>' +
        '<div style="color:var(--muted);">' + oLoDre.poss + ' poss</div></div></div>'
      ) : '';

      const mxcRec = (mxcRow.W != null) ? (mxcRow.W + '-' + mxcRow.L) : '';
      const oppRec = (oppRow.W != null) ? (oppRow.W + '-' + oppRow.L) : '';

      el.innerHTML =
        '<div style="font-size:9.5px;color:var(--muted);text-align:center;margin-bottom:3px;">' + esc(fg.date) + ' &middot; <b>MXC ' + esc(fg.loc) + ' ' + esc(oppAb) + '</b></div>' +
        '<div style="display:flex;align-items:center;justify-content:space-around;padding:2px 0 6px;border-bottom:1px solid var(--border);margin-bottom:5px;">' +
          '<div style="text-align:center;">' +
            '<img src="' + glLogoUrl('MXC') + '" title="MXC" style="width:38px;height:38px;object-fit:contain;display:block;margin:0 auto;">' +
            (mxcRec ? '<div style="font-size:9px;font-weight:700;color:var(--blue);">' + mxcRec + '</div>' : '') +
          '</div>' +
          '<div style="font-family:Oswald,sans-serif;font-weight:700;font-size:17px;color:var(--muted);">VS</div>' +
          '<div style="text-align:center;">' +
            oppImgHtml +
            (oppRec ? '<div style="font-size:9px;font-weight:700;color:var(--muted);">' + oppRec + '</div>' : '') +
          '</div>' +
        '</div>' +
        '<div style="font-family:Oswald,sans-serif;font-size:11px;font-weight:700;text-align:center;color:#000;margin-bottom:3px;">Season-Long Four Factors</div>' +
        '<table style="width:100%;border-collapse:collapse;margin-bottom:5px;">' +
          '<thead><tr style="background:#f1f5f9;font-family:Oswald,sans-serif;font-size:9.5px;color:#000;">' +
            '<th style="padding:3px 4px;border:1px solid #d4d4d4;"></th>' +
            '<th style="padding:3px 4px;border:1px solid #d4d4d4;">EFG%</th>' +
            '<th style="padding:3px 4px;border:1px solid #d4d4d4;">TOV%</th>' +
            '<th style="padding:3px 4px;border:1px solid #d4d4d4;">OR%</th>' +
            '<th style="padding:3px 4px;border:1px solid #d4d4d4;">FTR%</th>' +
          '</tr></thead>' +
          '<tbody>' +
'MXC' + mxcCells + '' +
'' + esc(oppAb) + '' + oppCells + '' +
'' +
'' +
'' + esc(oppAb) + ' SEASON-LONG DUOS' +
'' +
renderOppDuo('Best Duo', oppBestD, true) + renderOppDuo('Worst Duo', oppWorstD, false) +
'' +
((oHiHtml || oLoHtml) ? ('' + oHiHtml + oLoHtml + '') : '');
return;
}
